public class MinHeap {

    private int[] heap;
    private int size;

    public MinHeap(int cap){
        heap = new int[cap];
        size = 0;
    }

    public void insert(int val){
        if (size == heap.length){
            System.out.println("Heap is full");
            return;
        }
        heap[size] = val;
        swim(size);
        size++;
    }

    private void swim(int idx) {
        while (idx > 0 && heap[idx] < heap[(idx - 1) / 2]) {
            int parent = (idx - 1) / 2;
            swap(idx, parent);
            idx = parent;
        }
    }

    public int extractMin(){
        if (size == 0){
            System.out.println("Heap is empty");
        }
        int min = heap[0];
        heap[0] = heap[--size];
        sink(0);
        return min;
    }

    private void sink(int idx) {
        while (2 * idx + 1 < size) {
            int leftChild = 2 * idx + 1;
            int rightChild = leftChild + 1;
            int smallest = leftChild;

            if (rightChild < size && heap[rightChild] < heap[leftChild]) {
                smallest = rightChild;
            }

            if (heap[idx] <= heap[smallest]) {
                break;
            }
            swap(idx, smallest);
            idx = smallest;
        }
    }

    private void swap(int i, int j) {
        int temp = heap[i];
        heap[i] = heap[j];
        heap[j] = temp;
    }

    public void sort() {
        int length = size;
        for (int i = size - 1; i > 0; i--) {
            swap(0, i);
            size--;
            sink(0);
        }
        size = length;
    }

    public void printHeap() {
        for (int i = 0; i < size; i++) {
            System.out.print(heap[i] + " ");
        }
        System.out.println();
    }

    public static void main(String[] args) {
        MinHeap minHeap = new MinHeap(10);
        minHeap.insert(3);
        minHeap.insert(1);
        minHeap.insert(6);
        minHeap.insert(5);
        minHeap.insert(2);
        minHeap.insert(4);
        minHeap.printHeap();
        minHeap.sort();
        minHeap.printHeap();
    }
}

import java.util.Arrays;
import java.util.Scanner;

public class Taskscheduler {

    private int[] heap;
    private int size;
    private int capacity;
    private boolean isMinHeap;

    public Taskscheduler(int capacity, boolean isMinHeap) {
        this.capacity = capacity;
        this.heap = new int[capacity];
        this.size = 0;
        this.isMinHeap = isMinHeap;
    }

    public void insert(int value) {
        if (size == capacity) {
            System.out.println("Heap is full");
            return;
        }
        heap[size] = value;
        swim(size++);
    }

    private void swim(int index) {
        while (index > 0 && compare(heap[index], heap[parent(index)])) {
            swap(index, parent(index));
            index = parent(index);
        }
    }

    public int extractRoot() {
        if (size == 0) {
            System.out.println("Heap is empty");
        }
        int root = heap[0];
        heap[0] = heap[--size];
        sink(0);
        return root;
    }

    private void sink(int index) {
        int child;
        while ((child = leftChild(index)) < size) {
            if (rightChild(index) < size && compare(heap[rightChild(index)], heap[child])) {
                child = rightChild(index);
            }
            if (!compare(heap[child], heap[index])) {
                break;
            }
            swap(index, child);
            index = child;
        }
    }

    private int parent(int index) { return (index - 1) / 2; }
    private int leftChild(int index) { return 2 * index + 1; }
    private int rightChild(int index) { return 2 * index + 2; }
    private void swap(int i, int j) {
        int temp = heap[i];
        heap[i] = heap[j];
        heap[j] = temp;
    }
    private boolean compare(int a, int b) { return isMinHeap ? a < b : a > b; }

    public void displayHeap() {
        System.out.println("Heap:");
        System.out.println(Arrays.toString(Arrays.copyOf(heap, size)));
    }

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        System.out.print("Enter capacity: ");
        int capacity = sc.nextInt();
        System.out.print("Is Min Heap (true/false): ");
        boolean isMinHeap = sc.nextBoolean();

        Taskscheduler heap = new Taskscheduler(capacity, isMinHeap);
        System.out.print("Enter number of elements: ");
        int n = sc.nextInt();
        System.out.println("Enter elements:");
        for (int i = 0; i < n; i++) {
            heap.insert(sc.nextInt());
        }
        heap.displayHeap();
        System.out.println("Extracted root: " + heap.extractRoot());
        heap.displayHeap();
        sc.close();
    }
}

import java.util.Arrays;
import java.util.Scanner;

public class LargestK{

    private int[] heap;
    private int size;
    private int capacity;

    public LargestK(int capacity) {
        this.capacity = capacity;
        this.heap = new int[capacity];
        this.size = 0;
    }

    public void insert(int value) {
        if (size == capacity) {
            System.out.println("Heap is full");
            return;
        }
        heap[size] = value;
        swim(size++);
    }

    private void swim(int index) {
        while (index > 0 && heap[index] > heap[parent(index)]) {
            swap(index, parent(index));
            index = parent(index);
        }
    }

    public int extractMax() {
        if (size == 0) {
            System.out.println("Heap is empty");
        }
        int max = heap[0];
        heap[0] = heap[--size];
        sink(0);
        return max;
    }

    private void sink(int index) {
        int largerChild;
        while ((largerChild = leftChild(index)) < size) {
            int right = rightChild(index);
            if (right < size && heap[right] > heap[largerChild]) {
                largerChild = right;
            }
            if (heap[index] >= heap[largerChild]) break;
            swap(index, largerChild);
            index = largerChild;
        }
    }

    private int parent(int index) { return (index - 1) / 2; }
    private int leftChild(int index) { return 2 * index + 1; }
    private int rightChild(int index) { return 2 * index + 2; }
    private void swap(int i, int j) {
        int temp = heap[i];
        heap[i] = heap[j];
        heap[j] = temp;
    }

    public static int[] findTopK(int[] nums, int k) {
        LargestK maxHeap = new LargestK(nums.length);
        for (int num : nums) {
            maxHeap.insert(num);
        }
        int[] result = new int[k];
        for (int i = 0; i < k; i++) {
            result[i] = maxHeap.extractMax();
        }
        return result;
    }

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        System.out.print("Enter number of elements: ");
        int n = sc.nextInt();
        int[] nums = new int[n];
        System.out.println("Enter elements:");
        for (int i = 0; i < n; i++) {
            nums[i] = sc.nextInt();
        }
        System.out.print("Enter value of k: ");
        int k = sc.nextInt();
        int[] result = findTopK(nums, k);
        System.out.println("Top " + k + " largest elements: " + Arrays.toString(result));
        sc.close();
    }
}
