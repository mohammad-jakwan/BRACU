    public static Integer[] decryptMatrix( Integer[][] matrix ){
        int rows = matrix.length;
        int cols = matrix[0].length;
        Integer[] count = new Integer[cols];
        for (int i = 0; i < cols; i++){
            count[i] = 0;
            for (int j = 0; j < rows; j++) {
                count[i] += matrix[j][i];
            }
        }
        Integer[] output = new Integer[cols - 1];
        for (int i = 0; i < cols - 1; i++) {
            output[i] = count[i + 1] - count[i];
        }
        return output;
    }

    public static Integer rowRotation( Integer examWeek, String[][] matrix ){
            int rows = matrix.length;
            int rotation = examWeek % rows;

            for (int r = 0; r < rotation; r++) {
                String[] lastRow = matrix[rows - 1];
                for (int i = rows - 1; i > 0; i--) {
                    matrix[i] = matrix[i - 1]; 
                }
                matrix[0] = lastRow; 
            }
        
            int aaRow = -1;
            for (int i = 0; i < rows; i++) {
                for (int j = 0; j < matrix[i].length; j++){
                    if (matrix[i][j] == "AA") { 
                        aaRow = i + 1; 
                        break;
                    }
                }
                if (aaRow != -1){
                    break; 
                }
            }
            return aaRow;
    }

    public static void playGame( Integer[][] arena ){
            int rows = arena.length;
            int cols = arena[0].length;
            int points = 0;

            for (int i = 0; i < rows; i++) {
                for (int j = 0; j < cols; j++) {
                    if (arena[i][j] % 50 == 0 && arena[i][j] != 0) {
                        if (i > 0 && arena[i - 1][j] == 2) { 
                            points++;
                        }
                        if (i < rows - 1 && arena[i + 1][j] == 2) { 
                            points++;
                        }
                        if (j > 0 && arena[i][j - 1] == 2) { 
                            points++;
                        }
                        if (j < cols - 1 && arena[i][j + 1] == 2) { 
                            points++;
                        }
                    }
                }
            }
