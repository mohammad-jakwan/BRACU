public class Borrower{
  public static int book_count[] = {3, 3, 3};
  public static String book_name[] = {"Pather Panchali", "Durgesh Nandini", "Anandmath"};
  public static String borrowed[]= {null,null,null};
  public String name;
  public int count;
   
  
  public static void bookStatus(){
    System.out.println("Available Boooks:");
    for (int i=0; i<3; i++){
        System.out.println(book_name[i] + " : " + book_count[i]);
      }
    }
  
  
  public Borrower(String x){
    name = x;
  }
  
  public void borrowBook(String x){
    for (int i=0; i<book_name.length; i++){
      if (book_name[i]==x){
        if (book_count[i]>0){
            borrowed[count]= x;
            book_count[i]-=1;
            count++;
        }
        else{
        System.out.println("This book is not available.");
        }
      }
    }
}
                             
  
  public void borrowerDetails(){
    System.out.println("Name: " + name);
    System.out.println("Books Borrowed:");
    for (int i = 0; i<3; i++){
      if (borrowed[i]!= null){
        System.out.println(borrowed[i]);
      }
    }
  }
  
  public static int remainingBooks( String x) {
    int temp=0;
    for (int i=0; i<book_name.length; i++){
      if (book_name[i]==x){
        temp= book_count[i];
      }
    }
    return temp;
  }
  
}

public class Cargo{
  public static double capacity = 10;
  public static int id;
  public Boolean flag = false;
  public String content;
  public double weight;
  
  
  public Cargo(String x,double y ){
    content=x;
    weight = y;
    id++;
    flag = false;
  }
    public static double capacity(){
    return capacity;
  }
  
  public void load(){ 
    if (capacity>=weight){
      flag = true;
      capacity-= weight;
      System.out.println("Cargo "+ id +" loaded for transport.");
    }
    else{
      System.out.println("Cannot load cargo, exceeds weight capacity.");
    }
  }
  
  public void unload(){
    capacity+=weight;
    System.out.println("Cargo"+id+" unloaded.");
    flag = false;
  }
  
    public void details(){
      System.out.print("Cargo ID : " + id + ", ");
      System.out.print("Contents: " +content + ", ");
      System.out.print("Weight: "+ weight + ", ");
      System.out.println("Loaded: "+ flag);
    }
}
