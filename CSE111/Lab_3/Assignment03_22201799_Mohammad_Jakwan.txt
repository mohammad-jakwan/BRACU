import java.util.Scanner;

public class L3_T1{
  public static void main (String [] args){
    
    Scanner sc = new Scanner(System.in);
    System.out.println("PLease enter a string:");
    String input = sc.nextLine();
    String empty = "";
    String empty2 = "";
  
  int mid =input.length()/2;
  if(input.length()%2==0){
    for (int i =0; (i<(input.length()- mid)); i++){
      char a= input.charAt(i);
      empty=empty+a;
    }
    
    for (int j = (input.length()-1); j>= mid; j --){
      char b=input.charAt(j);
      empty2=empty2+b;
    }
  }
    else{
      for (int i =0; (i<(input.length()- mid-1)); i++){
        char a= input.charAt(i);
        empty=empty+a;
      }
      for (int j = (input.length()-1); j> mid; j --){
        char b=input.charAt(j);
        empty2=empty2+b;
    }
  
    }
    if (empty.equals(empty2)){
      System.out.println("Palindrome");
    }
    else{
      System.out.println("Not a palindrome");
}
}
}

public class L3_T4 {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        while (true) {
            int count = 1;
            System.out.println("Enter a string:");
            String str = sc.nextLine();
            String d = "";
            int leng = str.length();
            boolean e = false;
            for (int i = 0; i < leng; i++) {
                char ch = str.charAt(i);
                count = 0;
                for (int j = 0; j < leng; j++) {
                    if (ch == str.charAt(j) && !d.contains("" + ch)) {
                        count++;
                    }
                }
                if (count > 1) {
                    System.out.println("'" + ch + "' has been counted " + count + " times in the word " + str);
                    d += ch;
                    e = true;
                }
            }

            if (e == false) {
                System.out.println("You entered " + str);
                break;
            }
        }
    }
}

            arr[i] = sc.nextInt();
        }
        int[] newarr = new int[N];
        int count = 0;
        newarr[0] = arr[0];
        int index=1;
        for (int i = 1; i < N; i++) {
            if (arr[i] != arr[i - 1]) {
                newarr[index] = arr[i];
                index++;
            } else {
                count++;
            }
        }
        System.out.print("New Array: ");
        for (int i = 0; i < index; i++) {
            System.out.print(newarr[i] + " ");
        }
        System.out.println();
        System.out.println("Removed elements: " + count);
    }
}
