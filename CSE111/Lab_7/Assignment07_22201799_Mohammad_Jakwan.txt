public class Shape2D {
  private int side;
  private int breadth;
  private int length;
  private Boolean flag = true;
    
    public Shape2D(int x){
      length=x;
      System.out.println("A Square has been created with length: "+length);
    }
    
    public Shape2D(int x, int y){
      length=x;
      breadth = y;
      System.out.println("A Rectangle has been created with length: "+length+ " and breadth: "+breadth);
    }
    
    public Shape2D(int x, int y, String s){
      if(s!= null){
        flag = false;
      }
      length=x;
      breadth = y;
      System.out.println("A Tringle has been created with height: "+length+ " and base: "+breadth);
      
    }

    public Shape2D(int x, int y, int s){
      length=x;
      breadth = y;
      side = s;
      System.out.println("A Tringle has been created with the following sides: "+ length+ ", " + breadth + ", "+ side);
    }
    
    public void area(){
      if (side== 0 && breadth == 0 && length>0){
        System.out.println("The area of the Square is: " + (double)length*length);
      }
      if (flag = false ){
        System.out.println("The area of the Triangle is: " + (double)0.5*length*breadth);
      }
      if (side == 0 && length>0 && breadth>0){
        System.out.println("The area of the Rectangle is: " + (double)length*breadth);
      }
      if (side> 0 && breadth >0 && length>0){
        double p = (side+length+breadth)/2;
        double a = Math.sqrt(p* (p-side) * (p-length) * (p-breadth));
        System.out.println("The area of the Triangle is: " + a);
      }
      }
      

}

public class Student{
    private int id;
    private double cgpa;
    private String[] courses;
    private int count;
    private boolean flag;

    public Student(int x){
        id =x;
        courses = new String[4];
        count = 0;
        flag = false;
    }

    public Student(int x, double y){
        id = x;
        cgpa = y;
        courses = new String[4]; 
        count =0;
        flag = true;
    }

    public void setID(int x){
        id = x;
    }

    public void setCG(double x){
        cgpa =x ;
        flag = true;
    }

    public void addCourse(String course){
        if (!flag){
            System.out.println("Failed to add " + course);
            System.out.println("Set CG first");
        }
        if (cgpa < 3.0 && count >= 3){
            System.out.println("Failed to add " + course);
            System.out.println("CG is low. Can't add more than 3 courses.");   
        }
        if (count >= 4){
            System.out.println("Failed to add " + course);
            System.out.println("Maximum 4 courses allowed.");
        }
        courses[count] = course;
        count++;
    }

    public void addCourse(String[] arr){
      for (int i = 0; i < arr.length; i++){
        addCourse(arr[i]);
      }
    }

    public void rmAllCourse(){
        for (int i = 0; i < count; i++){
            courses[i] = null;
        }
        count = 0;
    }

    public void showAdvisee() {
        System.out.println("Student ID: " + id + ", CGPA: " + cgpa);
        if (count == 0){
            System.out.println("No courses added.");
        } 
        else{
            System.out.print("Added courses are: ");
            for (int i = 0; i < count; i++) {
                System.out.print(courses[i] + " ");
            }
        }
    }
}

public class ABCServer {
    private String serverName;
    private int capacity;
    private String[] members;
    private String[] roles;
    private int count;

    public ABCServer(){
        serverName = "Default";
        capacity = 10;
        members = new String[capacity];
        roles = new String[capacity];
        count = 0;
    }

    public ABCServer(String nm, int cp){
        serverName = nm;
        capacity = cp;
        members = new String[capacity];
        roles = new String[capacity];
        count = 0;
    }

    public void addMembers(String m){
        if (count >= capacity){
            System.out.println("Sorry, maximum capacity exceeded!");
        }
        members[count] = m;
        roles[count] = "Rising Hero";
        count++;
        System.out.println("Rising Hero is added.");
    }

    public void addMembers(String m, String r) {
        if (count >=capacity) {
            System.out.println("Sorry, maximum capacity exceeded!");
            return;
        }
        members[count] = m;
        roles[count] = r;
        count++;
        System.out.println(r + " is added.");
    }

    public void details(){
        System.out.println("Server Name: " + serverName);
        System.out.println("Member Capacity: " + capacity);
        System.out.println("Total Members: " + count);
        System.out.println("Members:");
        for (int i = 0; i < count; i++) {
            System.out.println("Name:Role --> " + members[i] + ":" + roles[i]);
        }
    }
}
