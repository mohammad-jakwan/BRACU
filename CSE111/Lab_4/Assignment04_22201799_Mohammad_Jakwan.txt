public class Bird{
  String name;
  int height=0;
  
  public void flyUp(int h){
    height+= h;
    System.out.println(name+ " has flown up "  + h +" feet");
  }
  public void flyDown(int i){
    
    if (i>height){
      System.out.println(name+ " cannot fly down "  + i +" feet");
    }
    if(i<height){
      System.out.println(name+ " has flown down "  + i +" feet");
      height-=i;
    }
    if(i==height){
      System.out.println(name+ " has flown down "  + i +" feet and landed");
      height-=i;
    }
  }
  public void makeNoise(){
    if (name.equals("Parrot")){
      System.out.println("Squawk");
    }
       
    else{
      System.out.println("Squee");
    }
  }
}

public class CellPhone{
  String model= "unknown";
  String[] contacts = new String[3]; 
  int count =0;
  
  public void storeContact(String s){
    if (count<3){
      contacts[count]= s;
      count++;
      System.out.println("Contact Stored");
      
    }
    else{
      System.out.println("Memory full. New contact can't be stored.");
    }
  }
    
    public void printDetails(){
      System.out.println("Phone Model "+model);
      System.out.println("Contacts Stored " + count );
      if (count>0){
      System.out.println("Stored Contacts:");
      
      for (int i =0; i <count; i++){
        System.out.println(contacts[i]);
      }
      }
    }
}
