    int sum= 0;
    int max = -99999;
    int min = 99999;
    double counter = 0.0;
    int count=0;
    
    while (count<10){

      int input = sc.nextInt();
      counter+=1;
      sum+=input;   
        if (input<min){
        min = input;
        }
        if (input>max){
        max = input;
        }
        count+=1;
      }
    System.out.println("Sum = "+sum);
    System.out.println("Minimum = "+min);
    System.out.println("Maximum = "+max);
    System.out.println("Average = " + (sum/counter));
  }
  }

import java.util.Scanner;
public class task210{

    public static void main(String[] args) {
        Scanner sc=new Scanner(System.in);
        System.out.println("Please enter an amount of money:");
        int amount=sc.nextInt();
        int[] notes = {500, 100, 50, 10, 5, 1};
        
        

        

        for (int i = 0; i < notes.length; i++) {
            int count=0;
            while (amount >= notes[i]) {
                amount -= notes[i];
                count++;
            }
            if(count!=0){
                System.out.println(count+" "+notes[i]+"'s note");
            }
        }

    
    }
}
